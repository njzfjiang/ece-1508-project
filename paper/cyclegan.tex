        \node[box, fill=blue!8, minimum width=5.0cm, minimum height=1.2cm] (init)

        % Down Blocks
        \node[box, fill=red!5, minimum width=4.5cm, minimum height=0.9cm] (down1)

        \node[box, fill=red!5, minimum width=4.5cm, minimum height=0.9cm] (up1)
